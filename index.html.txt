    <script>
        // Web Audio Synthesizer for retro dynamic SFX
        class SoundSystem {
            constructor() {
                this.ctx = null;
                this.muted = false;
            }

            init() {
                if (!this.ctx) {
                    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
                }
                if (this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }

            playCoin() {
                if (this.muted) return;
                this.init();
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(987.77, now); // B5
                osc.frequency.exponentialRampToValueAtTime(1318.51, now + 0.12); // E6
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start(now);
                osc.stop(now + 0.12);
            }

            playHit() {
                if (this.muted) return;
                this.init();
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, now);
                osc.frequency.exponentialRampToValueAtTime(40, now + 0.3);
                gain.gain.setValueAtTime(0.4, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start(now);
                osc.stop(now + 0.3);
            }

            playPowerup() {
                if (this.muted) return;
                this.init();
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(300, now);
                osc.frequency.linearRampToValueAtTime(800, now + 0.25);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start(now);
                osc.stop(now + 0.25);
            }

            playJump() {
                if (this.muted) return;
                this.init();
                const now = this.ctx.currentTime;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(220, now);
                osc.frequency.exponentialRampToValueAtTime(440, now + 0.15);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start(now);
                osc.stop(now + 0.15);
            }
        }

        const soundSystem = new SoundSystem();
    </script>

        // Dynamic DLI Coin Texture Generator Canvas
        function createDLICoinTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 256;
            canvas.height = 256;
            const ctx = canvas.getContext('2d');

            // Gold Radial Background
            const grad = ctx.createRadialGradient(128, 128, 10, 128, 128, 128);
            grad.addColorStop(0, '#fef08a');
            grad.addColorStop(0.5, '#eab308');
            grad.addColorStop(1, '#a16207');
            ctx.fillStyle = grad;
            ctx.beginPath();
            ctx.arc(128, 128, 120, 0, Math.PI * 2);
            ctx.fill();

            // Inner Ring Rim
            ctx.strokeStyle = '#fef08a';
            ctx.lineWidth = 12;
            ctx.beginPath();
            ctx.arc(128, 128, 100, 0, Math.PI * 2);
            ctx.stroke();

            // Embossed "DLI" Text
            ctx.font = '900 80px "Fredoka One", sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';

            // Text Shadow / Emboss Effect
            ctx.fillStyle = '#713f12';
            ctx.fillText('DLI', 131, 131);
            ctx.fillStyle = '#ffffff';
            ctx.fillText('DLI', 128, 128);

            const texture = new THREE.CanvasTexture(canvas);
            texture.needsUpdate = true;
            return texture;
        }

        // Update Textures when users upload photos
        function applyCustomTexture(type, imageSrc) {
            const loader = new THREE.TextureLoader();
            loader.load(imageSrc, (texture) => {
                texture.encoding = THREE.sRGBEncoding;
                customTextures[type] = texture;

                if (type === 'player' && playerHeadMesh) {
                    playerHeadMesh.material.map = texture;
                    playerHeadMesh.material.needsUpdate = true;
                } else if (type === 'police' && policeGroup) {
                    policeGroup.children[1].material.map = texture; // Head mesh
                    policeGroup.children[1].material.needsUpdate = true;
                } else if (type === 'dog' && dogGroup) {
                    dogGroup.children[0].material.map = texture; // Body/Head mesh
                    dogGroup.children[0].material.needsUpdate = true;
                }
            });
        }

        function createTrackSegment(zPos) {
            const segment = new THREE.Group();
            segment.position.z = zPos;

            // Ground Track Bed
            const groundGeo = new THREE.PlaneGeometry(14, 60);
            const groundMat = new THREE.MeshStandardMaterial({ color: 0x334155, roughness: 0.9 });
            const ground = new THREE.Mesh(groundGeo, groundMat);
            ground.rotation.x = -Math.PI / 2;
            ground.receiveShadow = true;
            segment.add(ground);

            // 3 Railway Tracks & Wooden Sleepers
            LANES.forEach(x => {
                for (let i = -28; i < 30; i += 2.5) {
                    const sleeper = new THREE.Mesh(
                        new THREE.BoxGeometry(2.0, 0.08, 0.4),
                        new THREE.MeshStandardMaterial({ color: 0x451a03 })
                    );
                    sleeper.position.set(x, 0.04, i);
                    segment.add(sleeper);
                }

                // Steel Rails
                [-0.7, 0.7].forEach(railOffset => {
                    const rail = new THREE.Mesh(
                        new THREE.BoxGeometry(0.12, 0.15, 60),
                        new THREE.MeshStandardMaterial({ color: 0x94a3b8, metalness: 0.8 })
                    );
                    rail.position.set(x + railOffset, 0.12, 0);
                    segment.add(rail);
                });
            });

            // Side Walls / Environment
            const wallMat = new THREE.MeshStandardMaterial({ color: 0x1e293b });
            const wallL = new THREE.Mesh(new THREE.BoxGeometry(1, 8, 60), wallMat);
            wallL.position.set(-8, 4, 0);
            segment.add(wallL);

            const wallR = new THREE.Mesh(new THREE.BoxGeometry(1, 8, 60), wallMat);
            wallR.position.set(8, 4, 0);
            segment.add(wallR);

            scene.add(segment);
            return segment;
        }

        function buildInitialTrack() {

                trackSegments.push(segment);
            }
        }

        function spawnCoinCluster(zPos) {
            const lane = LANES[Math.floor(Math.random() * 3)];
            const patternLength = 4 + Math.floor(Math.random() * 4);

            for (let i = 0; i < patternLength; i++) {
                const coinGroup = new THREE.Group();

                // 3D Cylinder Round Coin Mesh with DLI Face Texture
                const coinGeo = new THREE.CylinderGeometry(0.55, 0.55, 0.14, 24);
                const coinMat = [
                    new THREE.MeshStandardMaterial({ color: 0xf59e0b, metalness: 0.9, roughness: 0.2 }), // Rim
                    new THREE.MeshStandardMaterial({ map: dliCoinTexture, roughness: 0.2 }), // Top DLI Face
                    new THREE.MeshStandardMaterial({ map: dliCoinTexture, roughness: 0.2 })  // Bottom DLI Face
                ];

                const coinMesh = new THREE.Mesh(coinGeo, coinMat);
                coinMesh.rotation.x = Math.PI / 2; // Face forward
                coinGroup.add(coinMesh);

                // Round Glowing Aura / Soft Blur Sprite Outer Mesh
                const auraGeo = new THREE.SphereGeometry(0.75, 16, 16);
                const auraMat = new THREE.MeshBasicMaterial({
                    color: 0xfacc15,
                    transparent: true,
                    opacity: 0.35,
                    wireframe: false
                });
                const auraMesh = new THREE.Mesh(auraGeo, auraMat);
                coinGroup.add(auraMesh);

                coinGroup.position.set(lane, 1.2, zPos - i * 3.5);
                scene.add(coinGroup);

                coinObjects.push({
                    group: coinGroup,
                    mesh: coinMesh,
                    active: true
                });
            }
        }

        function spawnPowerup(zPos) {
            if (Math.random() > 0.4) return;
            const lane = LANES[Math.floor(Math.random() * 3)];
            const types = ['jetpack', 'magnet', 'hoverboard', 'sneakers'];
            const chosenType = types[Math.floor(Math.random() * types.length)];

            const group = new THREE.Group();
            let color = 0x3b82f6;
            if (chosenType === 'jetpack') color = 0xef4444;
            if (chosenType === 'magnet') color = 0xeab308;
            if (chosenType === 'hoverboard') color = 0x10b981;

            const mesh = new THREE.Mesh(
                new THREE.BoxGeometry(0.8, 0.8, 0.8),
                new THREE.MeshStandardMaterial({ color: color, metalness: 0.8 })
            );
            mesh.position.y = 1.2;
            group.add(mesh);

            group.position.set(lane, 0, zPos);
            scene.add(group);

            powerupObjects.push({ group: group, type: chosenType, active: true });
        }

        function setupInputs() {
            window.addEventListener('keydown', (e) => {
                if (gameState !== GAME_STATES.PLAYING) return;

                if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') {
                    if (currentLane > 0) currentLane--;
                } else if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') {
                    if (currentLane < 2) currentLane++;
                } else if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W' || e.key === ' ') {
                    if (isGrounded) {
                        velocityY = JUMP_FORCE * (activePowerups.sneakers > 0 ? 1.45 : 1.0);
                        isGrounded = false;
                        soundSystem.playJump();
                    }
                } else if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') {
                    if (!isRolling) {
                        isRolling = true;
                        rollTimer = 0.6;
                        if (!isGrounded) velocityY = -20; // Fast drop
                    }
                }
            });

            // Mobile Touch Swipe Handling
            let touchStartX = 0;
            let touchStartY = 0;

            window.addEventListener('touchstart', (e) => {
                touchStartX = e.touches[0].clientX;
                touchStartY = e.touches[0].clientY;
            });

            window.addEventListener('touchend', (e) => {
                if (gameState !== GAME_STATES.PLAYING) return;
                const dx = e.changedTouches[0].clientX - touchStartX;
                const dy = e.changedTouches[0].clientY - touchStartY;

                if (Math.abs(dx) > Math.abs(dy)) {
                    // Horizontal swipe
                    if (dx > 30 && currentLane < 2) currentLane++;
                    if (dx < -30 && currentLane > 0) currentLane--;
                } else {
                    // Vertical swipe
                    if (dy < -30 && isGrounded) {
                        velocityY = JUMP_FORCE * (activePowerups.sneakers > 0 ? 1.45 : 1.0);
                        isGrounded = false;
                        soundSystem.playJump();
                    } else if (dy > 30) {
                        isRolling = true;
                        rollTimer = 0.6;
                        if (!isGrounded) velocityY = -20;
                    }
                }
            });
        }

        function handlePlayerHit() {
            if (invulnerableTime > 0) return; // Currently shielded

            soundSystem.playHit();
            lives--;
            updateLivesUI();

            if (lives <= 0) {
                // Game Over
                endGame();
            } else {
                // Grant 2.5 Seconds Invulnerability Shield & Knockback
                invulnerableTime = 2.5;
                chaserDistance = 0.2; // Inspector moves right behind player!

                // Trigger visual alert
                const toast = document.getElementById('status-toast');
                toast.innerText = `LIVES REMAINING: ${lives}`;
                toast.classList.remove('hidden');
                setTimeout(() => toast.classList.add('hidden'), 1800);
            }
        }

        function updateLivesUI() {
            const container = document.getElementById('lives-display');
            container.innerHTML = '';
            for (let i = 0; i < 3; i++) {
                const heart = document.createElement('i');
                if (i < lives) {
                    heart.className = 'fa-solid fa-heart text-red-500 text-2xl heart-active filter drop-shadow';
                } else {
                    heart.className = 'fa-solid fa-heart text-slate-600 text-2xl filter drop-shadow opacity-40';
                }
                container.appendChild(heart);
            }
        }

        let clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);

            const delta = Math.min(clock.getDelta(), 0.1);

            if (gameState === GAME_STATES.PLAYING) {
                updateGameplay(delta);
            }

            renderer.render(scene, camera);
        }

        function updateGameplay(delta) {
            // Speed progression
            gameSpeed = Math.min(BASE_SPEED + (score / 400), MAX_SPEED);
            score += Math.round(gameSpeed * delta * 2);
            document.getElementById('score-text').innerText = String(score).padStart(6, '0');

            // Manage Invulnerability
            if (invulnerableTime > 0) {
                invulnerableTime -= delta;
                playerGroup.visible = Math.floor(Date.now() / 100) % 2 === 0; // Flashing effect
            } else {
                playerGroup.visible = true;
            }

            // Powerups Timer Update
            Object.keys(activePowerups).forEach(key => {
                if (activePowerups[key] > 0) {
                    activePowerups[key] -= delta;
                }
            });

            // Smooth Lane Transition
            targetX = LANES[currentLane];
            playerGroup.position.x += (targetX - playerGroup.position.x) * 15 * delta;

            // Vertical Jump / Gravity Physics
            if (activePowerups.jetpack > 0) {
                playerY = 7.0; // Fly high
                jetpackFire.visible = true;
            } else {
                jetpackFire.visible = false;
                velocityY += GRAVITY * delta;
                playerY += velocityY * delta;

                if (playerY <= 0) {
                    playerY = 0;
                    velocityY = 0;
                    isGrounded = true;
                }
            }
            playerGroup.position.y = playerY;

            // Rolling scale down
            if (isRolling) {
                rollTimer -= delta;
                playerGroup.scale.set(1, 0.5, 1);
                if (rollTimer <= 0) {
                    isRolling = false;
                    playerGroup.scale.set(1, 1, 1);
                }
            }

            // Update Chaser Position
            chaserDistance = Math.min(chaserDistance + delta * 0.15, 1.0);
            const chaserZ = playerGroup.position.z + 2.2 + (1.0 - chaserDistance) * 3.5;
            policeGroup.position.set(playerGroup.position.x * 0.8, playerGroup.position.y, chaserZ);
            dogGroup.position.set(playerGroup.position.x * 0.8 + 0.9, playerGroup.position.y, chaserZ + 0.4);

            // Move Track Segments
            trackSegments.forEach(segment => {
                segment.position.z += gameSpeed * delta;
                if (segment.position.z > 60) {
                    segment.position.z -= 300; // Recycle to back
                    // Spawn new obstacles & coins on recycled segment
                    spawnObstacle(segment.position.z - 20);
                    spawnCoinCluster(segment.position.z - 10);
                    spawnPowerup(segment.position.z - 30);
                }
            });

            // Update & Collect DLI Coins
            coinObjects.forEach(coin => {
                if (!coin.active) return;
                coin.group.position.z += gameSpeed * delta;

                // Spin Coin
                coin.mesh.rotation.z += 3.0 * delta;

                // Magnet Powerup Attraction Logic
                if (activePowerups.magnet > 0 && coin.group.position.z > -25) {
                    coin.group.position.x += (playerGroup.position.x - coin.group.position.x) * 10 * delta;
                    coin.group.position.y += (playerGroup.position.y + 1.2 - coin.group.position.y) * 10 * delta;
                }

                // Collision Check
                const dist = playerGroup.position.distanceTo(coin.group.position);
                if (dist < 1.6) {
                    coin.active = false;
                    scene.remove(coin.group);
                    coinsCollected++;
                    document.getElementById('coin-text').innerText = coinsCollected;
                    soundSystem.playCoin();
                }

                // Cleanup offscreen coins
                if (coin.group.position.z > 15) {
                    coin.active = false;
                    scene.remove(coin.group);
                }
            });

            // Collision Detection with Obstacles
            obstacles.forEach(obs => {
                obs.group.position.z += gameSpeed * delta;

                // Check overlap
                const dx = Math.abs(playerGroup.position.x - obs.group.position.x);
                const dz = Math.abs(playerGroup.position.z - obs.group.position.z);
                const dy = playerGroup.position.y - obs.group.position.y;

                if (dx < obs.width * 0.45 && dz < obs.depth * 0.45 && dy < obs.height) {
                    if (activePowerups.hoverboard > 0) {
                        // Hoverboard saves player from hit once
                        activePowerups.hoverboard = 0;
                        invulnerableTime = 2.0;
                        soundSystem.playHit();
                    } else {
                        handlePlayerHit();
                    }
                }
            });

            // Powerup Pickup Collision
            powerupObjects.forEach(pow => {
                if (!pow.active) return;
                pow.group.position.z += gameSpeed * delta;

                if (playerGroup.position.distanceTo(pow.group.position) < 1.5) {
                    pow.active = false;
                    scene.remove(pow.group);
                    activePowerups[pow.type] = 10.0; // 10s active duration
                    soundSystem.playPowerup();
                }
            });

            // Camera follow smoothly
            camera.position.x += (playerGroup.position.x * 0.4 - camera.position.x) * 5 * delta;
            camera.position.y = 5 + playerY * 0.3;
        }

        function startGame() {
            lives = 3;
            score = 0;
            coinsCollected = 0;
            gameSpeed = BASE_SPEED;
            invulnerableTime = 0;
            currentLane = 1;
            playerGroup.position.set(0, 0, 0);

            updateLivesUI();
            document.getElementById('score-text').innerText = '000000';
            document.getElementById('coin-text').innerText = '0';

            // Clear previous items
            obstacles.forEach(o => scene.remove(o.group));
            coinObjects.forEach(c => scene.remove(c.group));
            powerupObjects.forEach(p => scene.remove(p.group));
            obstacles = [];
            coinObjects = [];
            powerupObjects = [];

            // Spawn fresh batch
            for (let z = -30; z > -250; z -= 40) {
                spawnObstacle(z);
                spawnCoinCluster(z + 15);
            }

            document.getElementById('start-modal').classList.add('hidden');
            document.getElementById('gameover-modal').classList.add('hidden');
            document.getElementById('pause-modal').classList.add('hidden');

            gameState = GAME_STATES.PLAYING;
            clock.start();
        }

        function endGame() {
            gameState = GAME_STATES.GAMEOVER;
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('subway_surfer_dli_highscore', highScore);
            }

            document.getElementById('final-score-text').innerText = score;
            document.getElementById('high-score-text').innerText = highScore;
            document.getElementById('final-coins-text').innerText = coinsCollected;
            document.getElementById('gameover-modal').classList.remove('hidden');
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // Setup File Uploader Listeners for Custom Avatars
        function handleFileUpload(inputEl, previewImgId, previewIconId, type) {
            inputEl.addEventListener('change', (e) => {
                const file = e.target.files[0];
                if (file) {
                    const reader = new FileReader();
                    reader.onload = function(evt) {
                        const src = evt.target.result;
                        if (previewImgId) {
                            const img = document.getElementById(previewImgId);
                            img.src = src;
                            img.classList.remove('hidden');
                            document.getElementById(previewIconId).classList.add('hidden');
                        }
                        applyCustomTexture(type, src);
                    };
                    reader.readAsDataURL(file);
                }
            });
        }
